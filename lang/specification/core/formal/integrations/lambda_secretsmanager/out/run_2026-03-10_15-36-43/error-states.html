# Lambda -> Secrets Manager Integration Formal Specification
# Models a Lambda function reading a secret from Secrets Manager during invocation.
# If the secret is scheduled for deletion (pending deletion window), the read fails
# and the invocation fails with a ResourceNotFoundException.
 
FUNC_IDS = ["func-1"]
SECRET_IDS = ["secret-1"]
INV_IDS = ["inv-1"]
 
 
action Init:
    func_status = {}            # func_status[fid] = "ACTIVE"
    secret_status = {}          # secret_status[sid] = "ACTIVE" | "PENDING_DELETION"
    inv_status = {}             # inv_status[iid] = "IN_PROGRESS" | "SUCCESS" | "FAILED"
    inv_func = {}               # inv_func[iid] = func_id
    inv_secret = {}             # inv_secret[iid] = secret_id (secret read during invocation)
 
 
# step: a Lambda function is deployed
# result: the function is ACTIVE
# guard: the function does not already exist
# guard_violation: the function already exists
atomic action DeployFunction:
    any fid in FUNC_IDS:
        if fid not in func_status:
            func_status[fid] = "ACTIVE"
 
 
# step: a secret is created in Secrets Manager
# result: the secret is ACTIVE and can be read by Lambda
# guard: the secret does not already exist
# guard_violation: the secret already exists
atomic action CreateSecret:
    any sid in SECRET_IDS:
        if sid not in secret_status:
            secret_status[sid] = "ACTIVE"
 
 
# step: a secret is scheduled for deletion
# result: the secret is PENDING_DELETION and will be unavailable to Lambda during the recovery window
# guard: the secret exists
# guard_violation: the secret does not exist
# guard: the secret is ACTIVE
# guard_violation: the secret is not ACTIVE
atomic action ScheduleSecretDeletion:
    any sid in SECRET_IDS:
        if sid in secret_status:
            if secret_status[sid] == "ACTIVE":
                secret_status[sid] = "PENDING_DELETION"
 
 
# step: the Lambda function is invoked
# result: the invocation is IN_PROGRESS
# guard: the function exists
# guard_violation: the function does not exist
# guard: the function is ACTIVE
# guard_violation: the function is not ACTIVE
# guard: an invocation slot is available
# guard_violation: no invocation slot is available
atomic action InvokeFunction:
    any fid in FUNC_IDS:
        if fid in func_status:
            if func_status[fid] == "ACTIVE":
                any iid in INV_IDS:
                    if iid not in inv_status or inv_status[iid] == "SUCCESS" or inv_status[iid] == "FAILED":
                        inv_status[iid] = "IN_PROGRESS"
                        inv_func[iid] = fid
 
 
# step: the Lambda function reads an ACTIVE secret and completes successfully
# result: the invocation is SUCCESS
# guard: an invocation is IN_PROGRESS
# guard_violation: no invocation is IN_PROGRESS
# guard: the secret exists and is ACTIVE
# guard_violation: the secret does not exist or is not ACTIVE
atomic action InvocationSucceeds:
    any iid in INV_IDS:
        if iid in inv_status:
            if inv_status[iid] == "IN_PROGRESS":
                any sid in SECRET_IDS:
                    if sid in secret_status:
                        if secret_status[sid] == "ACTIVE":
                            inv_status[iid] = "SUCCESS"
                            inv_secret[iid] = sid
 
 
# step: the Lambda function fails because the secret is pending deletion
# result: the invocation is FAILED with a ResourceNotFoundException
# guard: an invocation is IN_PROGRESS
# guard_violation: no invocation is IN_PROGRESS
# guard: the secret is PENDING_DELETION
# guard_violation: the secret is not pending deletion
atomic action InvocationFailsSecretUnavailable:
    any iid in INV_IDS:
        if iid in inv_status:
            if inv_status[iid] == "IN_PROGRESS":
                any sid in SECRET_IDS:
                    if sid in secret_status:
                        if secret_status[sid] == "PENDING_DELETION":
                            inv_status[iid] = "FAILED"
                            inv_secret[iid] = sid
 
 
# check: every IN_PROGRESS invocation references an ACTIVE Lambda function
always assertion InvocationRequiresActiveFunction:
    for iid in inv_status:
        if inv_status[iid] == "IN_PROGRESS":
            fid = inv_func[iid]
            if fid in func_status:
                if func_status[fid] != "ACTIVE":
                    return False
    return True
 
 
# check: every successful invocation read an ACTIVE secret
always assertion SuccessfulInvocationRequiresActiveSecret:
    for iid in inv_status:
        if inv_status[iid] == "SUCCESS":
            if iid in inv_secret:
                sid = inv_secret[iid]
                if sid in secret_status:
                    if secret_status[sid] != "ACTIVE":
                        return False
    return True
